cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(ExtraPaneLinuxUiChecks LANGUAGES CXX)
endif()
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
add_executable(extrapane-linux-window-check window_check.cpp)
target_compile_features(extrapane-linux-window-check PRIVATE cxx_std_20)
target_link_libraries(extrapane-linux-window-check PRIVATE Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
set_source_files_properties(../../apps/linux/ReceiverWindow.qml PROPERTIES QT_RESOURCE_ALIAS ReceiverWindow.qml)
set_source_files_properties(Video.qml PROPERTIES QT_RESOURCE_ALIAS Video.qml)
qt_add_resources(extrapane-linux-window-check ui PREFIX / FILES ../../apps/linux/ReceiverWindow.qml Video.qml)
enable_testing()
add_test(NAME linux-receiver-window COMMAND extrapane-linux-window-check)

# Reuse the project's real credential parser; transport is a deterministic fake.
if(EXTRAPANE_ACCESS_LIBRARY AND EXTRAPANE_CRYPTO_LIBRARY)
  add_executable(extrapane-linux-admission-check admission_check.cpp ../../apps/linux/admission.cpp ../../apps/linux/admission.hpp)
  target_compile_features(extrapane-linux-admission-check PRIVATE cxx_std_20)
  target_compile_definitions(extrapane-linux-admission-check PRIVATE EXTRAPANE_ADMISSION_TEST)
  target_include_directories(extrapane-linux-admission-check PRIVATE . ../../apps/linux ../../core/access/include)
  target_link_libraries(extrapane-linux-admission-check PRIVATE Qt6::Core "${EXTRAPANE_ACCESS_LIBRARY}" "${EXTRAPANE_CRYPTO_LIBRARY}")
  if(WIN32)
    set_property(TARGET extrapane-linux-admission-check PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
    target_link_libraries(extrapane-linux-admission-check PRIVATE advapi32 bcrypt)
  endif()
  add_test(NAME linux-admission-lifecycle COMMAND extrapane-linux-admission-check)
endif()
