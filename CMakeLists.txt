cmake_minimum_required(VERSION 3.28)
project(dm VERSION 0.13.0 LANGUAGES C CXX)

# Standalone diagnostic executable: no VC runtime installation on the test laptop.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This first native slice requires Windows and MSVC.")
endif()

add_library(dm_probe_protocol STATIC core/protocol/probe.cpp core/protocol/video_protocol.cpp core/protocol/input_protocol.cpp)
target_include_directories(dm_probe_protocol PUBLIC core/protocol/include)
target_compile_features(dm_probe_protocol PUBLIC cxx_std_20)

include(cmake/Dependencies.cmake)
add_library(dm_network STATIC platform/windows/network.cpp)
target_include_directories(dm_network PUBLIC platform/windows/include)
target_link_libraries(dm_network PUBLIC dm_probe_protocol ws2_32 iphlpapi bcrypt)
target_compile_definitions(dm_network PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 UNICODE _UNICODE)
add_executable(dm-net apps/net/main.cpp)
target_link_libraries(dm-net PRIVATE dm_network)

add_library(dm_security STATIC platform/windows/pairing.cpp platform/windows/tls.cpp)
target_link_libraries(dm_security PUBLIC dm_network mbedtls PRIVATE advapi32)
add_executable(dm-session apps/session/main.cpp)
target_link_libraries(dm-session PRIVATE dm_security)

add_library(dm_video STATIC platform/windows/video_capture.cpp platform/windows/monitor_capture.cpp platform/windows/video_stream.cpp platform/windows/video_preview.cpp platform/windows/h264.cpp platform/windows/video_gpu.cpp platform/windows/input_capture.cpp platform/windows/input_broker.cpp)
target_link_libraries(dm_video PUBLIC dm_security PRIVATE dm_display windowsapp d3d11 d3dcompiler dxgi dxguid windowscodecs ole32 oleaut32 mfplat mfuuid mf user32 wmcodecdspuuid)
add_executable(dm-view apps/view/main.cpp)
target_link_libraries(dm-view PRIVATE dm_video)

add_library(dm_display STATIC platform/windows/display.cpp)
target_include_directories(dm_display PUBLIC platform/windows/include)
target_compile_features(dm_display PUBLIC cxx_std_20)
target_compile_definitions(dm_display PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 UNICODE _UNICODE)
target_link_libraries(dm_display PRIVATE swdevice cfgmgr32 user32 advapi32)
add_executable(dm-display apps/display/main.cpp apps/display/pattern.cpp)
target_link_libraries(dm-display PRIVATE dm_display gdi32)

include(CTest)
if(BUILD_TESTING)
  add_executable(dm-input-tests tests/input_protocol_tests.cpp)
  target_link_libraries(dm-input-tests PRIVATE dm_probe_protocol)
  target_compile_options(dm-input-tests PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  add_test(NAME input-protocol COMMAND dm-input-tests)
  add_executable(dm-input-check tests/input_hardware_check.cpp)
  target_link_libraries(dm-input-check PRIVATE dm_video)
  target_compile_options(dm-input-check PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  add_executable(dm-tray-check tests/tray_check.cpp)
  target_link_libraries(dm-tray-check PRIVATE dm_display user32)
  add_executable(dm-window-check tests/preview_window_check.cpp)
  target_compile_features(dm-window-check PRIVATE cxx_std_20)
  target_link_libraries(dm-window-check PRIVATE user32)
  add_executable(dm-input-drive tests/input_drive.cpp)
  target_link_libraries(dm-input-drive PRIVATE dm_display)
  target_compile_options(dm-input-drive PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  add_executable(dm-protocol-tests tests/probe_tests.cpp)
  target_link_libraries(dm-protocol-tests PRIVATE dm_probe_protocol)
  add_test(NAME protocol COMMAND dm-protocol-tests)
  find_package(Python3 3.10 COMPONENTS Interpreter REQUIRED)
  add_test(NAME network-integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/network_integration.py" "$<TARGET_FILE:dm-net>")
  set_tests_properties(network-integration PROPERTIES TIMEOUT 40)
  add_test(NAME security-integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/security_integration.py" "$<TARGET_FILE:dm-session>")
  set_tests_properties(security-integration PROPERTIES TIMEOUT 40)
  add_executable(dm-video-tests tests/video_tests.cpp)
  target_link_libraries(dm-video-tests PRIVATE dm_video)
  target_compile_options(dm-video-tests PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  add_test(NAME video-protocol-and-codec COMMAND dm-video-tests)

foreach(target dm_probe_protocol dm_network dm_security dm_video dm_display dm-display dm-net dm-session dm-view)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
endforeach()
if(TARGET dm-protocol-tests)
  target_compile_options(dm-protocol-tests PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
endif()

# Lightweight native launchers for the existing engine. WinUI product shell remains separate.
foreach(app dm-host dm-receiver)
  add_executable(${app} WIN32 apps/desktop/main.cpp)
  target_link_libraries(${app} PRIVATE dm_security ole32 oleaut32 user32 gdi32)
  target_compile_options(${app} PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
endforeach()
target_compile_definitions(dm-host PRIVATE DM_HOST_APP)
target_link_options(dm-host PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")

# Brand shipped executables, including the native preview/helper processes.
enable_language(RC)
foreach(app dm-view dm-display dm-host dm-receiver)
  target_sources(${app} PRIVATE apps/winui/app.rc)
  target_include_directories(${app} PRIVATE assets/branding)
endforeach()

add_executable(SideScreen.Service WIN32 apps/service/main.cpp apps/winui/app.rc)
target_include_directories(SideScreen.Service PRIVATE assets/branding)
target_link_libraries(SideScreen.Service PRIVATE dm_security dm_display ole32 oleaut32 shell32)
target_link_options(SideScreen.Service PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
