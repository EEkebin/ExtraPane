cmake_minimum_required(VERSION 3.28)
project(ExtraPaneSharedUi LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Network Widgets)
add_subdirectory(../../core/session session)
add_library(extrapane_ui_model STATIC resource_model.cpp resource_model.hpp)
target_link_libraries(extrapane_ui_model PUBLIC extrapane_session Qt6::Core)
target_include_directories(extrapane_ui_model PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(extrapane-ui-check ../../tests/shared_ui_check.cpp)
target_link_libraries(extrapane-ui-check PRIVATE extrapane_connection_model extrapane_ui_model Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
set_source_files_properties(ui/Main.qml PROPERTIES QT_RESOURCE_ALIAS Main.qml)
set_source_files_properties(ui/ResourcePicker.qml PROPERTIES QT_RESOURCE_ALIAS ResourcePicker.qml)
set_source_files_properties(ui/ClientConnections.qml PROPERTIES QT_RESOURCE_ALIAS ClientConnections.qml)
set_source_files_properties(ui/AppIcon.qml PROPERTIES QT_RESOURCE_ALIAS AppIcon.qml)
set_source_files_properties(ui/IconButton.qml PROPERTIES QT_RESOURCE_ALIAS IconButton.qml)
set_source_files_properties(../../assets/branding/icon.png PROPERTIES QT_RESOURCE_ALIAS extrapane.png)
qt_add_resources(extrapane-ui-check ui PREFIX / FILES ui/Main.qml ui/ResourcePicker.qml ui/ClientConnections.qml ui/AppIcon.qml ui/IconButton.qml ../../assets/branding/icon.png)
enable_testing()
add_library(extrapane_connection_model STATIC connection_model.cpp connection_model.hpp)
target_link_libraries(extrapane_connection_model PUBLIC Qt6::Core Qt6::Network)
target_include_directories(extrapane_connection_model PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(extrapane-connection-model-check ../../tests/connection_model_check.cpp)
target_link_libraries(extrapane-connection-model-check PRIVATE extrapane_connection_model)
target_compile_features(extrapane-connection-model-check PRIVATE cxx_std_20)
add_test(NAME connection-model COMMAND extrapane-connection-model-check)
add_test(NAME shared-ui-model COMMAND extrapane-ui-check)
set_tests_properties(shared-ui-model PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
if(WIN32)
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/release_version.py"
    --header "${CMAKE_CURRENT_BINARY_DIR}/ExtraPaneVersion.h" COMMAND_ERROR_IS_FATAL ANY)
  add_executable(ExtraPane WIN32 windows_main.cpp windows_controller.cpp windows_controller.hpp windows_connection_runtime.cpp windows_connection_runtime.hpp ../winui/app.rc)
  target_include_directories(ExtraPane PRIVATE ../../assets/branding "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_definitions(ExtraPane PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
  target_link_libraries(ExtraPane PRIVATE extrapane_connection_model extrapane_ui_model Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Widgets bcrypt advapi32 shell32)
  set_source_files_properties(ui/WindowsVideo.qml PROPERTIES QT_RESOURCE_ALIAS Video.qml)
  set_source_files_properties(../../assets/branding/icon.png PROPERTIES QT_RESOURCE_ALIAS extrapane.png)
  qt_add_resources(ExtraPane app_ui PREFIX / FILES ui/Main.qml ui/ResourcePicker.qml ui/ClientConnections.qml ui/AppIcon.qml ui/IconButton.qml ui/WindowsVideo.qml ../../assets/branding/icon.png)
  add_executable(extrapane-controller-fake ../../tests/shared_controller_fake.cpp)
  add_executable(extrapane-controller-check ../../tests/shared_controller_check.cpp windows_controller.cpp windows_controller.hpp windows_connection_runtime.cpp windows_connection_runtime.hpp)
  target_include_directories(extrapane-controller-check PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_definitions(extrapane-controller-check PRIVATE EXTRAPANE_CONTROLLER_TEST_CHILDREN NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
  target_link_libraries(extrapane-controller-check PRIVATE extrapane_connection_model extrapane_ui_model Qt6::Network bcrypt advapi32 shell32)
  set_target_properties(extrapane-controller-check PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/controller-check")
  add_dependencies(extrapane-controller-check extrapane-controller-fake)
  foreach(helper ExtraPane.Session.exe ExtraPane.Resources.exe ExtraPane.Microphone.exe ExtraPane.Speakers.exe ExtraPane.Usb.exe dm-view.exe)
    add_custom_command(TARGET extrapane-controller-check POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:extrapane-controller-fake>" "$<TARGET_FILE_DIR:extrapane-controller-check>/${helper}")
  endforeach()
  add_test(NAME shared-controller-lifecycle COMMAND extrapane-controller-check)
endif()
