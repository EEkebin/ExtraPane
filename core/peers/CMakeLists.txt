cmake_minimum_required(VERSION 3.28)
project(ExtraPanePeers LANGUAGES CXX)
add_library(extrapane_peers STATIC peers.cpp store.cpp rpc.cpp media_bridge.cpp)
target_include_directories(extrapane_peers PUBLIC include)
target_compile_features(extrapane_peers PUBLIC cxx_std_20)
if(WIN32)
  target_link_libraries(extrapane_peers PRIVATE bcrypt crypt32 advapi32 ws2_32)
elseif(TARGET crypto)
  target_compile_definitions(extrapane_peers PRIVATE BORINGSSL_PREFIX=EXTRAPANE)
  target_link_libraries(extrapane_peers PRIVATE crypto)
else()
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  target_link_libraries(extrapane_peers PRIVATE OpenSSL::Crypto)
endif()
option(EXTRAPANE_PEERS_TESTS "Build isolated peer catalog/policy checks" OFF)
if(EXTRAPANE_PEERS_TESTS)
  enable_testing()
  add_executable(extrapane-peers-check ../../tests/peers_check.cpp)
  target_link_libraries(extrapane-peers-check PRIVATE extrapane_peers)
  add_test(NAME peers-wire-policy COMMAND extrapane-peers-check)
  add_executable(extrapane-peers-store-check ../../tests/peers_store_check.cpp)
  target_link_libraries(extrapane-peers-store-check PRIVATE extrapane_peers)
  add_test(NAME peers-private-store COMMAND extrapane-peers-store-check)
  add_executable(extrapane-peer-rpc-check ../../tests/peer_rpc_check.cpp)
  target_link_libraries(extrapane-peer-rpc-check PRIVATE extrapane_peers)
  add_test(NAME peer-rpc-flow-control COMMAND extrapane-peer-rpc-check)
  add_executable(extrapane-peers-media-check ../../tests/peers_media_bridge_check.cpp)
  target_link_libraries(extrapane-peers-media-check PRIVATE extrapane_peers)
  add_test(NAME peers-loopback-media COMMAND extrapane-peers-media-check)
endif()
