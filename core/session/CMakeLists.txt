cmake_minimum_required(VERSION 3.28)
project(ExtraPaneSession LANGUAGES CXX)
add_library(extrapane_session STATIC session.cpp server.cpp wire.cpp channel.cpp negotiation.cpp media.cpp camera_protocol.cpp microphone_protocol.cpp usb.cpp usb_tunnel.cpp)
target_include_directories(extrapane_session PUBLIC include)
target_compile_features(extrapane_session PUBLIC cxx_std_20)
option(EXTRAPANE_SESSION_TESTS "Build isolated resource lifecycle checks" OFF)
if(EXTRAPANE_SESSION_TESTS)
  enable_testing()
  add_executable(extrapane-session-checks ../../tests/session_check.cpp)
  target_link_libraries(extrapane-session-checks PRIVATE extrapane_session)
  add_test(NAME resource-lifecycle COMMAND extrapane-session-checks)
  add_executable(extrapane-channel-checks ../../tests/channel_check.cpp)
  target_link_libraries(extrapane-channel-checks PRIVATE extrapane_session)
  add_test(NAME resource-channel COMMAND extrapane-channel-checks)
  add_executable(extrapane-media-checks ../../tests/media_check.cpp)
  target_link_libraries(extrapane-media-checks PRIVATE extrapane_session)
  add_test(NAME resource-media COMMAND extrapane-media-checks)
  add_executable(extrapane-camera-protocol-checks ../../tests/camera_protocol_check.cpp)
  target_link_libraries(extrapane-camera-protocol-checks PRIVATE extrapane_session)
  add_test(NAME camera-protocol COMMAND extrapane-camera-protocol-checks)
  add_executable(extrapane-usb-checks ../../tests/usb_check.cpp)
  target_link_libraries(extrapane-usb-checks PRIVATE extrapane_session)
  add_test(NAME usb-ownership COMMAND extrapane-usb-checks)
  add_executable(extrapane-usb-tunnel-checks ../../tests/usb_tunnel_check.cpp)
  target_link_libraries(extrapane-usb-tunnel-checks PRIVATE extrapane_session)
  add_test(NAME usb-tunnel COMMAND extrapane-usb-tunnel-checks)

  add_test(NAME microphone-protocol COMMAND extrapane-microphone-protocol-checks)
