enable_language(OBJCXX)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
add_executable(ExtraPane MACOSX_BUNDLE main.mm transport.mm media.mm media_check.mm)
target_link_libraries(ExtraPane PRIVATE dm_probe_protocol "-framework AppKit" "-framework AVFoundation" "-framework CoreMedia" "-framework CoreVideo" "-framework VideoToolbox" "-framework QuartzCore" "-framework Network" "-framework Security")
target_compile_options(ExtraPane PRIVATE -fobjc-arc -Wall -Wextra)
set_target_properties(ExtraPane PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in")
if(DEFINED ENV{EXTRAPANE_VERSION})
  set(EXTRAPANE_MAC_VERSION "$ENV{EXTRAPANE_VERSION}")
else()
  file(STRINGS "${PROJECT_SOURCE_DIR}/VERSION" EXTRAPANE_MAC_VERSION LIMIT_COUNT 1)
endif()
set_target_properties(ExtraPane PROPERTIES MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}" MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}")
target_compile_definitions(ExtraPane PRIVATE EXTRAPANE_MAC_VERSION="${EXTRAPANE_MAC_VERSION}")
enable_testing()
add_test(NAME macos-protocol-media-input COMMAND ExtraPane --self-test)
