set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
include(${PROJECT_SOURCE_DIR}/cmake/MbedTLS.cmake)
add_executable(ExtraPane MACOSX_BUNDLE main.mm peer_runtime.mm peer_display.mm peer_workspace.mm peer_discovery.cpp transport.mm media.mm media_check.mm display.mm host.mm host_tls.cpp host_media.mm host_input.mm admission.mm resource_session.mm resource_capture.mm resource_ui.mm)

target_compile_definitions(ExtraPane PRIVATE EXTRAPANE_MAC_VERSION="${EXTRAPANE_MAC_VERSION}")
